====================================================================
5. API GATEWAY: GET /coupons_poc on the "coupons" API  (scripts/t1_provision_coupons_endpoint.sh)
====================================================================

bash scripts/t1_provision_coupons_endpoint.sh

# Commands inside the script:
API_ID=$(awslocal apigateway get-rest-apis --region us-east-1 --query "items[?name==\`coupons\`].id" --output text)
ROOT_RESOURCE_ID=$(awslocal apigateway get-resources --rest-api-id ${API_ID} --query 'items[?path==`/`].id' --output text --region us-east-1)
awslocal apigateway create-resource --region us-east-1 --rest-api-id ${API_ID} --parent-id ${ROOT_RESOURCE_ID} --path-part "coupons_poc"
COUPONS_POC_RESOURCE_ID=$(awslocal apigateway get-resources --region us-east-1 --rest-api-id ${API_ID} --query "items[?path==\`/coupons_poc\`].id" --output text)
awslocal apigateway put-method --region us-east-1 --rest-api-id ${API_ID} --resource-id ${COUPONS_POC_RESOURCE_ID} --http-method GET --authorization-type "NONE"
COUPONS_LIST_LAMBDA_ARN=$(awslocal lambda list-functions --region us-east-1 --query "Functions[?FunctionName==\`coupons_list\`].FunctionArn" --output text)
awslocal apigateway put-integration --region us-east-1 --rest-api-id ${API_ID} --resource-id ${COUPONS_POC_RESOURCE_ID} --http-method GET --type AWS_PROXY --integration-http-method POST --uri arn:aws:apigateway:us-east-1:lambda:path/2015-03-31/functions/${COUPONS_LIST_LAMBDA_ARN}/invocations --passthrough-behavior WHEN_NO_MATCH
awslocal lambda add-permission --region us-east-1 --function-name "coupons_list" --statement-id "apigateway-get-coupons_poc" --action "lambda:InvokeFunction" --principal "apigateway.amazonaws.com" --source-arn "arn:aws:execute-api:us-east-1:000000000000:${API_ID}/*/GET/coupons_poc"
awslocal apigateway create-deployment --region us-east-1 --rest-api-id ${API_ID} --stage-name local

# Verify - call the endpoint
API_ID=$(awslocal apigateway get-rest-apis --query "items[?name==\`coupons\`].id" --output text)
curl -i "http://${API_ID}.execute-api.localhost.localstack.cloud:4566/local/coupons_poc"
curl -s "http://${API_ID}.execute-api.localhost.localstack.cloud:4566/local/coupons_poc" > output/coupons_poc_response.json
