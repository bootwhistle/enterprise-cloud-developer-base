5. API GATEWAY: GET /coupons_poc on the "coupons" API  (scripts/t1_provision_coupons_endpoint.sh)
====================================================================

bash scripts/t1_provision_coupons_endpoint.sh

# Commands inside the script:
API_ID=$(awslocal apigateway get-rest-apis --region us-east-1 --query "items[?name==\`coupons\`].id" --output text)
ROOT_RESOURCE_ID=$(awslocal apigateway get-resources --rest-api-id ${API_ID} --query 'items[?path==`/`].id' --output text --region us-east-1)
awslocal apigateway create-resource --region us-east-1 --rest-api-id ${API_ID} --parent-id ${ROOT_RESOURCE_ID} --path-part "coupons_poc"
COUPONS_POC_RESOURCE_ID=$(awslocal apigateway get-resources --region us-east-1 --rest-api-id ${API_ID} --query "items[?path==\`/coupons_poc\`].id" --output text)
awslocal apigateway put-method --region us-east-1 --rest-api-id ${API_ID} --resource-id ${COUPONS_POC_RESOURCE_ID} --http-method GET --authorization-type "NONE"
COUPONS_LIST_LAMBDA_ARN=$(awslocal lambda list-functions --region us-east-1 --query "Functions[?FunctionName==\`coupons_list\`].FunctionArn" --output text)
awslocal apigateway put-integration --region us-east-1 --rest-api-id ${API_ID} --resource-id ${COUPONS_POC_RESOURCE_ID} --http-method GET --type AWS_PROXY --integration-http-method POST --uri arn:aws:apigateway:us-east-1:lambda:path/2015-03-31/functions/${COUPONS_LIST_LAMBDA_ARN}/invocations --passthrough-behavior WHEN_NO_MATCH
awslocal lambda add-permission --region us-east-1 --function-name "coupons_list" --statement-id "apigateway-get-coupons_poc" --action "lambda:InvokeFunction" --principal "apigateway.amazonaws.com" --source-arn "arn:aws:execute-api:us-east-1:000000000000:${API_ID}/*/GET/coupons_poc"
awslocal apigateway create-deployment --region us-east-1 --rest-api-id ${API_ID} --stage-name local

# Verify - call the endpoint
API_ID=$(awslocal apigateway get-rest-apis --query "items[?name==\`coupons\`].id" --output text)
curl -i "http://${API_ID}.execute-api.localhost.localstack.cloud:4566/local/coupons_poc"
curl -s "http://${API_ID}.execute-api.localhost.localstack.cloud:4566/local/coupons_poc" > output/coupons_poc_response.json

1. DYNAMODB TABLE: coupons  (scripts/t2_create_dynamodb_table.sh)
   Capacity (provisioned, 1 KB average item):
     Reads : 50 eventually consistent reads/s. 1 RCU = 2 eventually
             consistent reads/s of up to 4 KB -> 50 / 2 = 25 RCU
     Writes: 10 writes/s. 1 WCU = 1 write/s of up to 1 KB -> 10 WCU
====================================================================

bash scripts/t2_create_dynamodb_table.sh

# Commands inside the script:
awslocal dynamodb create-table --table-name "coupons" --attribute-definitions AttributeName=coupon_id,AttributeType=S --key-schema AttributeName=coupon_id,KeyType=HASH --provisioned-throughput ReadCapacityUnits=25,WriteCapacityUnits=10
awslocal dynamodb wait table-exists --table-name "coupons"
awslocal dynamodb put-item --table-name "coupons" --item '{"coupon_id":{"S":"319326de971100158f6fa0daeec978fba1f17e4b"}, ...}'
#   (sample coupon from samples/t2_coupons_get_by_id_response.json - full item in the script)

====================================================================
4b. GLOBAL SECONDARY INDEX on provider_id + campaign_id
    (scripts/t2_add_index_to_dynamodb_table.sh)
====================================================================

bash scripts/t2_add_index_to_dynamodb_table.sh

# Commands inside the script:
awslocal dynamodb update-table --table-name "coupons" --attribute-definitions AttributeName=provider_id,AttributeType=S AttributeName=campaign_id,AttributeType=S --global-secondary-index-updates '[{"Create":{"IndexName":"provider_id-campaign_id-index","KeySchema":[{"AttributeName":"provider_id","KeyType":"HASH"},{"AttributeName":"campaign_id","KeyType":"RANGE"}],"Projection":{"ProjectionType":"ALL"},"ProvisionedThroughput":{"ReadCapacityUnits":25,"WriteCapacityUnits":10}}}]'
awslocal dynamodb wait table-exists --table-name "coupons"

# Verify - query by provider and campaign through the index (no table scan)
awslocal dynamodb query --table-name coupons --index-name provider_id-campaign_id-index --key-condition-expression "provider_id = :p AND campaign_id = :c" --expression-attribute-values '{":p":{"S":"FAIRMONT"},":c":{"S":"CHRISTMAS2022"}}'

# LocalStack 2.3.2 only: it routes a request to the FIRST matching resource in
# creation order, so GET /coupons/to_secure was sent to coupons_get_by_id
# (/coupons/{id} with id = "to_secure") and returned 500. Real API Gateway
# prefers the literal path. This recreates /coupons/{id} last (same GET/PUT
# integrations) so the literal /coupons/* paths are matched first.
bash scripts/localstack_fix_route_order.sh

# The seeded users (password field is the bcrypt hash, never the plain text)
awslocal dynamodb scan --table-name users

====================================================================
1. LAMBDA: coupons_get_token  (scripts/t3_update_coupons_get_token_lambda.sh)
   POST /token  body {"username": "...", "password": "..."}
   - reads the user from the "users" table
   - bcrypt.compare(password, stored hash)   (hash = bcrypt, 12 rounds)
   - signs a JWT (HS256) with the secret 0HXc4w5NEzA61HkV, expires in 3600 s
   200 {"token": "...", "expiresIn": 3600}
   403 wrong password or unknown user | 400 missing fields / bad JSON
   Libraries: bcryptjs (pure JS bcrypt), jsonwebtoken
====================================================================

npm install --prefix ./lambda/coupons_get_token
npm test --prefix ./lambda/coupons_get_token

bash scripts/t3_update_coupons_get_token_lambda.sh

# Commands inside the script (only production dependencies are packaged):
mkdir -p build/coupons_get_token && cp lambda/coupons_get_token/{index.js,package.json,package-lock.json} build/coupons_get_token/
npm ci --omit=dev --prefix build/coupons_get_token
cd build/coupons_get_token && zip -r ../../coupons_get_token.zip index.js node_modules && cd ../..
awslocal lambda update-function-code --function-name "coupons_get_token" --zip-file fileb://coupons_get_token.zip
awslocal lambda wait function-updated-v2 --function-name "coupons_get_token"

# Verify through API Gateway
API_ID=$(awslocal apigateway get-rest-apis --query "items[?name==\`coupons\`].id" --output text)
